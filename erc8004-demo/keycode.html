① IdentityRegistry身份注册表 —— 智能体的"数字营业执照"

event Registered(uint256 indexed agentId, string agentURI, address indexed owner);

// 注册：铸造 ERC-721，返回 agentId —— 链上只存 URI 指针
function register(string agentURI) external returns (uint256 agentId);
function register(string agentURI, MetadataEntry[] calldata metadata) external returns (uint256 agentId);

// 身份信息可更新 —— 履历随身份保留，URI 换了历史不丢
function setAgentURI(uint256 agentId, string calldata newURI) external;

// 保留的收款钱包字段：变更需 EIP-712 签名验证，防止冒充
function setAgentWallet(uint256 agentId, address newWallet, uint256 deadline, bytes calldata signature) external;
讲解要点：register() 只接收一个字符串指针（agentURI），注册文件本体在 IPFS/HTTPS 链下——这就是"链上存指针、链下存内容"。agentId 是 ERC-721 NFT，谁持有它谁拥有这个身份和全部历史。

② ReputationRegistry声誉注册表 —— 只存信号，不存评分

// 提交反馈：带符号定点数 int128 + 小数位，可正可负；证据文件在链下
function giveFeedback(uint256 agentId,
                     int128  value,          // 例: 87 表示 87 分，-32 表示 -3.2%（配合 decimals）
                     uint8   valueDecimals,  // 例: 2 → 实际含义 0.87
                     string  calldata tag1, tag2,      // 维度标签: quality / uptime / tradingYield…
                     string  calldata feedbackURI,     // 链下证据文件指针
                     bytes32 feedbackHash) external;

// 汇总查询：clientAddresses 为必填 —— 强制按客户端过滤，抑制女巫刷分
function getSummary(uint256 agentId, address[] calldata clientAddresses,
                   string tag1, string tag2)
                   external view returns (uint64 count, int128 summaryValue, uint8 valueDecimals);

function revokeFeedback(uint256 agentId, uint64 feedbackIndex) external;
讲解要点：协议刻意不定义"平均分"或"信任排名"——只标准化原始信号的存取。市场按完成速度加权、保险公司按纠纷历史算风险，各自消费同一份数据（可组合性）。getSummary 强制要求传入 clientAddresses 过滤，是规范里内建的女巫攻击缓解设计。

③ ValidationRegistry验证注册表 —— 高风险输出的"独立审计"

// 请求验证 → 验证者（TEE / zkML / 质押重执行）离线检查 → 链上应答 0-100 分
function validationRequest(address validatorAddress, uint256 agentId,
                          string calldata requestURI, bytes32 requestHash) external;

function validationResponse(bytes32 requestHash,
                           uint8   response,      // 0-100 的验证结果
                           string  calldata responseURI, bytes32 responseHash,
                           string  calldata tag) external;

// 按 requestHash 查询某次验证的状态；或按 agentId 聚合统计
function getValidationStatus(bytes32 requestHash) external view
  returns (address validatorAddress, uint256 agentId, uint8 response, uint256 lastUpdate);

function getSummary(uint256 agentId, address[] calldata validatorAddresses, string tag)
  external view returns (uint64 count, uint8 averageResponse);
讲解要点：验证模型是"可插拔"的——协议只记录请求和应答，不规定验证者怎么验证。安全性随风险价值升级：低风险交互看声誉即可，大额操作才请 TEE/zkML 独立验证（响应 0-100，链上任何人可查）。

⑤ 节奏建议的课堂展示顺序

  1. 先讲结构：用本页三个代码块讲"三大注册表各管什么"——身份（NFT）、声誉（信号）、验证（0-100 应答）。
  2. 再跑真流程：切到 AgentHub（node server.js），注册 → 下单 → 反馈，让抽象接口落到能看到的事件。
  3. 最后上真链：打开 Etherscan 的已验证合约或你自己的注册交易（sepolia/register.cjs 产出），证明"刚才的一切就是这笔交易"。

提示：讲解每段代码时，重点让学生注意 giveFeedback 的 int128 + tag 设计和 getSummary 的强制地址过滤——这两处最能体现"协议只做信任原语、不做应用决策"的设计哲学。
